23 smoothCutoffValue(const double x, const double r, const double d)
24 {
25 const double y = 1.0 - d * (x - r) / r;
26 const double f1_y = (y <= 0.0) ? 0.0 : exp(-1.0 / y);
27 const double omy = 1.0 - y;
28 const double f1_1my = (omy <= 0.0) ? 0.0 : exp(-1.0 / omy);
29 return f1_y / (f1_y + f1_1my);

32 DFTEFE_HOST_DEVICE_FUNC double
33 smoothCutoffDerivative(const double x,
34 const double r,
35 const double d,
36 const double tolerance)
37 {
38 const double y = 1.0 - d * (x - r) / r;
39 if (fabs(y) < tolerance || fabs(1.0 - y) < tolerance)
40 return 0.0;
41 const double f1_y = (y <= 0.0) ? 0.0 : exp(-1.0 / y);
42 const double omy = 1.0 - y;
43 const double f1_1my = (omy <= 0.0) ? 0.0 : exp(-1.0 / omy);
44 const double f1Der_y = f1_y / (y * y);
45 const double f1Der_1my = f1_1my / (omy * omy);
46 const double denom = f1_y + f1_1my;
47 const double f2Der =
48 (f1Der_y * f1_1my + f1_y * f1Der_1my) / (denom * denom);
49 return f2Der * (-d / r);
50 }

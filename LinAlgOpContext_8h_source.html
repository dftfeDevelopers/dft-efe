126#if defined(DFTEFE_WITH_DEVICE_LANG_SYCL)
127 // SYCL queues (unlike CUDA/HIP streams) have no implicit ordering
128 // relationship with one another, so the default-stream/non-default-
129 // stream mutual exclusivity that CUDA/HIP get for free from the driver
130 // (stream 0 implicitly synchronizes with every other blocking stream)
131 // has to be enforced here explicitly for callers that go through
132 // getBlasStream()/getDeviceBlasHandle()/getBlasStreamsVec()/
133 // getDeviceBlasHandlesVec().
134 enum class ActiveStreamSide
135 {
136 Default,
137 NonDefault
138 };
139
140 inline static ActiveStreamSide d_activeStreamSide =
141 ActiveStreamSide::Default;
142
143 static void
144 enforceDefaultStreamExclusivity()
145 {
146 if (d_activeStreamSide == ActiveStreamSide::NonDefault)
147 {
148 for (auto &stream : d_streams)
149 utils::deviceStreamSynchronize(stream);
150 d_activeStreamSide = ActiveStreamSide::Default;
151 }
152 }
153
154 static void
155 enforceNonDefaultStreamExclusivity()
156 {
157 if (d_activeStreamSide == ActiveStreamSide::Default)
158 {
159 utils::deviceStreamSynchronize(d_defaultStream);
160 d_activeStreamSide = ActiveStreamSide::NonDefault;
161 }
162 }
163#endif
